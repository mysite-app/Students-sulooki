قواعد أمان Firebase Firestore المقترحة

انسخ هذه القواعد إلى تبويب Rules في لوحة تحكم Firebase لحماية بيانات الطلاب وسرية الحالات: 

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /students/{studentId} {
      allow read, write: if request.auth != null;
    }
    match /cases/{caseId} {
      allow read, write: if request.auth != null;
    }
    match /settings/{docId} {
      allow read, write: if request.auth != null;
    }
  }
}
